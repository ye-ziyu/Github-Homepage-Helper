### 1.2 部署架构

#### 方案 A：纯前端架构（MVP）
- 前端部署在 Vercel/Netlify
- 所有 API 调用直接从浏览器发起
- 无需后端服务器
- Token 存储在 localStorage

#### 方案 B：前后端分离架构（推荐）
- 前端：Vercel/Netlify
- 后端：Railway/Render/Fly.io
- 数据库：Supabase/PostgreSQL
- Redis 缓存：可选

-- 用户配置表
CREATE TABLE user_configs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    config_key VARCHAR(50) NOT NULL,
    config_value JSONB NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, config_key)
);

-- 生成历史表
CREATE TABLE generation_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    generation_type VARCHAR(20) NOT NULL, -- 'bio', 'avatar', 'readme'
    input_config JSONB NOT NULL,
    output_result TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 同步历史表
CREATE TABLE sync_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,

    target_repo VARCHAR(100),
    status VARCHAR(20) NOT NULL, -- 'success', 'failed'
    error_message TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX idx_users_github_id ON users(github_id);
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_user_configs_user_id ON user_configs(user_id);
CREATE INDEX idx_generation_history_user_id ON generation_history(user_id);
CREATE INDEX idx_sync_history_user_id ON sync_history(user_id);

// 缓存策略
async function getUserInfo(username: string): Promise<UserInfo> {
  const cacheKey = CACHE_KEYS.USER_INFO(username);

  // 尝试从缓存获取
  const cached = await redis.get(cacheKey);
  if (cached) {
    return JSON.parse(cached);
  }

  // 从数据库获取
  const user = await db.users.findOne({ where: { username } });

  // 写入缓存
  await redis.setex(cacheKey, CACHE_TTL.USER_INFO, JSON.stringify(user));

  return user;
}

// 批处理请求
async function fetchMultipleRepos(repos: string[]): Promise<Repo[]> {
  const batchSize = 10;
  const results = [];

  for (let i = 0; i < repos.length; i += batchSize) {
    const batch = repos.slice(i, i + batchSize);
    const batchResults = await Promise.all(
      batch.map(repo => githubApi.getRepo(repo))
    );
    results.push(...batchResults);
  }

  return results;
}

  externalApiCalls: new Map<string, number>(),
};

// 记录 API 响应时间
function trackApiCall(endpoint: string, duration: number) {
  const times = performanceMetrics.apiResponseTime.get(endpoint) || [];
  times.push(duration);
  performanceMetrics.apiResponseTime.set(endpoint, times);

  // 只保留最近 100 次记录
  if (times.length > 100) {
    times.shift();
  }
}

// 计算平均响应时间
function getAverageResponseTime(endpoint: string): number {
  const times = performanceMetrics.apiResponseTime.get(endpoint) || [];
  if (times.length === 0) return 0;

  const sum = times.reduce((a, b) => a + b, 0);
  return sum / times.length;
}
